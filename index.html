Código Arduino C++ para ESP32 (LM335 + Servo)

Copia y carga este código en tu **ESP32** usando Arduino IDE. Configurado con los tópicos y puertos requeridos: 

#include <WiFi.h>
#include <PubSubClient.h>
#include <ESP32Servo.h>

// Credenciales WiFi
const char* ssid = "Maestros";
const char* password = "Docentes_2025";

// Servidor Broker MQTT (Puerto Estándar 1883)
const char* mqtt_server = "test.mosquitto.org";
const int mqtt_port = 1883;

// Tópicos asignados al TP
const char* TOPIC_TEMP  = "etusiemens/Marcenaro/temperatura";
const char* TOPIC_SERVO = "etusiemens/Discoteca/servo";

// Pines de Conexión
const int PIN_SERVO = 18; // Pin de Señal del Servo
const int PIN_LM335 = 34; // Entrada Analógica del Sensor LM335

WiFiClient espClient;
PubSubClient client(espClient);
Servo miServo;

unsigned long ultimoEnvio = 0;

void callback(char* topic, byte* payload, unsigned int length) {
  String mensaje = "";
  for (unsigned int i = 0; i < length; i++) {
    mensaje += (char)payload[i];
  }

  Serial.print("Comando recibido en [");
  Serial.print(topic);
  Serial.print("]: ");
  Serial.println(mensaje);

  // Mapeo: '1' -> Abrir (90°), '0' -> Cerrar (0°)
  if (mensaje == "1") {
    miServo.write(90); // Mueve a 90° (Abrir)
    Serial.println("Servo a 90° (Abierto)");
  } else if (mensaje == "0") {
    miServo.write(0);  // Mueve a 0° (Cerrar)
    Serial.println("Servo a 0° (Cerrado)");
  }
}

void setup_wifi() {
  delay(10);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Conectado!");
}

void reconnect() {
  while (!client.connected()) {
    String clientId = "ESP32Client-" + String(random(0xffff), HEX);
    if (client.connect(clientId.c_str())) {
      Serial.println("Conectado a MQTT Broker!");
      client.subscribe(TOPIC_SERVO);
    } else {
      delay(5000);
    }
  }
}

float leerTemperaturaLM335() {
  int valorADC = analogRead(PIN_LM335);
  float voltaje = (valorADC / 4095.0) * 3.3;
  float kelvin = voltaje * 100.0;
  return kelvin - 273.15; // Retorna Celsius
}

void setup() {
  Serial.begin(115200);
  miServo.attach(PIN_SERVO);
  miServo.write(0); // Posición inicial 0° (Cerrado)
  
  setup_wifi();
  client.setServer(mqtt_server, mqtt_port);
  client.setCallback(callback);
}

void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop();

  // Envía temperatura cada 3 segundos
  if (millis() - ultimoEnvio > 3000) {
    ultimoEnvio = millis();
    float tempC = leerTemperaturaLM335();
    String tempStr = String(tempC, 2);
    
    client.publish(TOPIC_TEMP, tempStr.c_str());
    Serial.print("Temperatura enviada: ");
    Serial.println(tempStr);
  }
}
